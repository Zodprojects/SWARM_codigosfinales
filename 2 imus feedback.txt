# -*- coding: utf-8 -*-
"""
Actuador lineal por CAN (dir/PWM) con opción de:
  - "seq": secuencia fija de (dir, pwm, dur)
  - "angle_pid": control de ángulo con MPU6050 (pitch/roll) como feedback

Requisitos:
  pip install python-can smbus2 adafruit-circuitpython-bno055
  Configurar can0 (p.ej., 500 kbps).
"""

from __future__ import annotations
import time, math, sys
from dataclasses import dataclass
from smbus2 import SMBus

# ======================= CONFIGURACIÓN =======================
@dataclass
class Config:
    # --- CAN ---
    can_channel: str = "can0"
    can_bitrate: int = 500_000

    # --- Actuador (IDs fuera de bandas ODrive) ---
    ACT_CAN_ID_CMD:  int = 0x321
    ACT_CAN_ID_DATA: int = 0x322  # si tu firmware la usa para telemetría
    act_refresh_s: float = 0.060

    # --- Modo: "seq" o "angle_pid" ---
    act_mode: str = "angle_pid"

    # --- Secuencia (solo en "seq") ---
    actuador_seq = [
        (1, 120, 1.5),
        ("wait", 0, 1.0),
        (2, 140, 1.5),
    ]

    # --- Control por ángulo (angle_pid) ---
    theta_ref_deg: float = 25.0    # consigna (grados)
    imu_dt: float = 0.02           # 50 Hz
    imu_alpha: float = 0.98        # filtro complementario
    imu_sign: float = +1.0         # invierte si tu eje está al revés (+1/-1)
    imu_axis: str = "pitch"        # "pitch" o "roll"
    # PID
    Kp: float = 3.0
    Ki: float = 0.0
    Kd: float = 0.25
    I_clamp: float = 0.6
    deadband_deg: float = 0.5
    # Salida PWM
    PWM_MAX: int = 255
    PWM_MIN_EFF: int = 70
    PWM_SLEW: int = 50
    # Límites de operación
    theta_min_deg: float = -5.0
    theta_max_deg: float = 65.0
    stall_timeout_s: float = 2.0
    # Telemetría
    telemetry_hz: float = 10.0
    show_both_axes: bool = True
    # (Opcional) BNO055 para comparar lecturas (no controla)
    use_bno055_telemetry: bool = False
    bno_addr: int = 0x28
    # (Opcional) duración máxima del control (None = infinito hasta Ctrl+C)
    run_seconds: float | None = None

# ======================= CAN =======================
class CanManager:
    def __init__(self, cfg: Config):
        import can
        self.bus = can.interface.Bus(channel=cfg.can_channel, bustype="socketcan")
    def send(self, arbitration_id: int, data: list[int]):
        import can
        msg = can.Message(arbitration_id=arbitration_id, data=data, is_extended_id=False)
        self.bus.send(msg)
    def shutdown(self):
        try: self.bus.shutdown()
        except Exception: pass

# ======================= Actuador (dir/PWM) =======================
class ActuadorDPWM:
    """TX: arbitration_id=cfg.ACT_CAN_ID_CMD, data=[dir, pwm]. Paro: [0,0]."""
    def __init__(self, cfg: Config, can_mgr: CanManager):
        self.cfg = cfg
        self.can = can_mgr
        self._pwm_last = 0
    def send(self, direccion: int, pwm: int):
        direccion = int(direccion) & 0xFF
        pwm = int(pwm) & 0xFF
        self.can.send(self.cfg.ACT_CAN_ID_CMD, [direccion, pwm])
    def stop(self):
        try: self.send(0, 0)
        except: pass
        self._pwm_last = 0
    def enviar_comando(self, direccion: int, pwm: int, duracion: float):
        t_end = time.perf_counter() + max(0.0, float(duracion))
        while time.perf_counter() < t_end:
            self.send(direccion, pwm)
            print(f"🟢 Actuador → dir:{direccion} pwm:{pwm}")
            time.sleep(self.cfg.act_refresh_s)
        self.stop()
        print("🛑 Actuador detenido.")

# ======================= IMUs =======================
# ---- MPU6050 (pitch/roll con filtro complementario) ----
MPU_ADDR = 0x68
PWR_MGMT_1 = 0x6B
ACCEL_XOUT_H = 0x3B
GYRO_XOUT_H  = 0x43
ACCEL_SCALE = 16384.0  # LSB/g
GYRO_SCALE  = 131.0    # LSB/(°/s)

class MPU6050:
    def __init__(self, bus=1, addr=MPU_ADDR, dt=0.02, alpha=0.98):
        self.bus = SMBus(bus)
        self.addr = addr
        self.dt = dt
        self.alpha = alpha
        self.bus.write_byte_data(self.addr, PWR_MGMT_1, 0x00)
        time.sleep(0.05)
        self.gyro_bias = self._calibrate_gyro()
        ax, ay, az, *_ = self._read_accel_gyro()
        self.pitch, self.roll = self._accel_to_angles(ax, ay, az)

    def _read_word(self, reg):
        hi = self.bus.read_byte_data(self.addr, reg)
        lo = self.bus.read_byte_data(self.addr, reg+1)
        val = (hi << 8) | lo
        if val >= 0x8000: val = -((65535 - val) + 1)
        return val

    def _read_accel_gyro(self):
        ax = self._read_word(ACCEL_XOUT_H)     / ACCEL_SCALE
        ay = self._read_word(ACCEL_XOUT_H+2)   / ACCEL_SCALE
        az = self._read_word(ACCEL_XOUT_H+4)   / ACCEL_SCALE
        gx = self._read_word(GYRO_XOUT_H)      / GYRO_SCALE
        gy = self._read_word(GYRO_XOUT_H+2)    / GYRO_SCALE
        gz = self._read_word(GYRO_XOUT_H+4)    / GYRO_SCALE
        return ax, ay, az, gx, gy, gz

    def _accel_to_angles(self, ax, ay, az):
        roll  = math.degrees(math.atan2(ay, az))
        pitch = math.degrees(math.atan2(-ax, math.sqrt(ay*ay + az*az)))
        return pitch, roll

    def _calibrate_gyro(self, samples=200):
        sx = sy = sz = 0.0
        for _ in range(samples):
            _, _, _, gx, gy, gz = self._read_accel_gyro()
            sx += gx; sy += gy; sz += gz
            time.sleep(0.002)
        return {'x': sx/samples, 'y': sy/samples, 'z': sz/samples}

    def read_pitch_roll(self):
        ax, ay, az, gx, gy, gz = self._read_accel_gyro()
        gx -= self.gyro_bias['x']; gy -= self.gyro_bias['y']
        pitch_gyro = self.pitch + gy * self.dt
        roll_gyro  = self.roll  + gx * self.dt
        pitch_acc, roll_acc = self._accel_to_angles(ax, ay, az)
        self.pitch = self.alpha * pitch_gyro + (1 - self.alpha) * pitch_acc
        self.roll  = self.alpha * roll_gyro  + (1 - self.alpha) * roll_acc
        return self.pitch, self.roll

# ---- BNO055 opcional (solo telemetría visual) ----
class BNO055_IMU:
    def __init__(self, addr=0x28):
        import board, adafruit_bno055
        i2c = board.I2C()
        self.sensor = adafruit_bno055.BNO055_I2C(i2c, address=addr)
    def read_pitch_roll(self):
        eul = self.sensor.euler  # (heading, roll, pitch)
        if eul is None: return None, None
        heading, roll, pitch = eul
        return pitch, roll

# ======================= Control por Ángulo =======================
class AnglePIDActuator:
    def __init__(self, cfg: Config, act: ActuadorDPWM):
        self.cfg = cfg
        self.act = act
        self.mpu = MPU6050(dt=cfg.imu_dt, alpha=cfg.imu_alpha)
        self.bno = BNO055_IMU(cfg.bno_addr) if cfg.use_bno055_telemetry else None
        self._ei = 0.0
        self._e_prev = 0.0
        self._pwm_prev = 0
        self._last_theta = None
        self._last_change_t = time.perf_counter()
        self._telemetry_last = 0.0

    def _pid(self, e, dt):
        self._ei += e*dt
        self._ei = max(-self.cfg.I_clamp, min(self.cfg.I_clamp, self._ei))
        de = (e - self._e_prev)/max(1e-3, dt)
        self._e_prev = e
        return self.cfg.Kp*e + self.cfg.Ki*self._ei + self.cfg.Kd*de

    def _pwm_from_u(self, u):
        mag = abs(u)
        if mag < self.cfg.deadband_deg:
            pwm = 0
        else:
            pwm = int(min(self.cfg.PWM_MAX, self.cfg.PWM_MIN_EFF + 5.0*mag))
        dp = max(-self.cfg.PWM_SLEW, min(self.cfg.PWM_SLEW, pwm - self._pwm_prev))
        pwm = self._pwm_prev + dp
        self._pwm_prev = pwm
        return pwm

    def _dir_from_u(self, u):
        return 1 if u > 0 else 2  # define: u>0 subir espejo (ajusta si invierte)

    def _print_status(self, theta_sel, pitch, roll, e, direccion, pwm):
        now = time.perf_counter()
        if now - self._telemetry_last >= (1.0 / max(1e-3, self.cfg.telemetry_hz)):
            self._telemetry_last = now
            eje = self.cfg.imu_axis
            if self.cfg.show_both_axes:
                line = (f"θref={self.cfg.theta_ref_deg:5.1f}°  "
                        f"θ{eje}={theta_sel:6.2f}°  "
                        f"pitch={pitch:6.2f}°  roll={roll:6.2f}°  "
                        f"e={e:6.2f}°  dir/pwm={direccion if pwm else 0}/{pwm:3d}")
            else:
                line = (f"θref={self.cfg.theta_ref_deg:5.1f}°  "
                        f"θ{eje}={theta_sel:6.2f}°  e={e:6.2f}°  "
                        f"dir/pwm={direccion if pwm else 0}/{pwm:3d}")
            sys.stdout.write("\r" + line + " " * 6)
            sys.stdout.flush()

    def run(self, max_seconds: float | None = None):
        print("▶️ Control ángulo (MPU6050) activo.")
        dt = self.cfg.imu_dt
        t_start = time.perf_counter()
        try:
            while True:
                if max_seconds is not None and (time.perf_counter() - t_start) >= max_seconds:
                    print("\n⏹️ Tiempo de control agotado.")
                    break

                t0 = time.perf_counter()
                pitch, roll = self.mpu.read_pitch_roll()
                if pitch is None or (pitch != pitch):
                    print("\n⚠️ IMU no válida. Paro.")
                    break

                theta_sel = pitch if self.cfg.imu_axis.lower() == "pitch" else roll
                theta_sel *= self.cfg.imu_sign

                # Límites de seguridad
                if theta_sel < self.cfg.theta_min_deg - 2 or theta_sel > self.cfg.theta_max_deg + 2:
                    print(f"\n⛔ θ fuera de rango ({theta_sel:.1f}°). Paro.")
                    break

                # Estancamiento (sin cambio significativo)
                if self._last_theta is not None and abs(theta_sel - self._last_theta) > 0.2:
                    self._last_change_t = t0
                self._last_theta = theta_sel
                if (t0 - self._last_change_t) > self.cfg.stall_timeout_s:
                    print("\n⛔ Estancado (sin movimiento). Paro.")
                    break

                # PID -> dir/pwm
                e = (self.cfg.theta_ref_deg - theta_sel)
                u = self._pid(e, dt)
                pwm = self._pwm_from_u(u)
                if pwm == 0:
                    self.act.stop()
                    direccion = 0
                else:
                    direccion = self._dir_from_u(u)
                    self.act.send(direccion, pwm)

                # Telemetría
                self._print_status(theta_sel, pitch, roll, e, direccion, pwm)

                # ciclo
                t_spent = time.perf_counter() - t0
                time.sleep(max(0.0, dt - t_spent))
        finally:
            self.act.stop()
            print("")

# ======================= MAIN =======================
def main():
    cfg = Config()
    canm = CanManager(cfg)
    act  = ActuadorDPWM(cfg, canm)
    print(f"▶️ Modo actuador: {cfg.act_mode}. Ctrl-C para parar.")
    try:
        if cfg.act_mode.lower() == "seq":
            for paso in cfg.actuador_seq:
                if isinstance(paso[0], str) and paso[0].lower() == "wait":
                    _, _, t_wait = paso
                    print(f"\n⏸️ Espera {t_wait:.2f}s")
                    time.sleep(max(0.0, float(t_wait)))
                    continue
                direccion, pwm, dur = paso
                act.enviar_comando(int(direccion), int(pwm), float(dur))
            act.stop()
        else:
            controller = AnglePIDActuator(cfg, act)
            controller.run(cfg.run_seconds)
    except KeyboardInterrupt:
        print("\n⛔ Interrupción usuario")
    except Exception as e:
        print("❌ FAULT:", e)
    finally:
        print("\n🛑 Parando…")
        try: act.stop()
        except: pass
        try: canm.shutdown()
        except: pass
        print("✅ Listo.")

if __name__ == "__main__":
    main()
